> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建 Job 拆分排产配置页面，实现从 Part Number 选择、BOM 全路径展开、逐层 Recipe 配置、Process 自动 Group 到总 Recipe 预览的完整闭环。

**Architecture:** 采用前后端分离架构，前端使用 Vue 3 + Element Plus 构建交互式树形配置界面，后端使用 Spring Boot 提供 BOM 展开、Recipe 筛选、Process Group 计算等 API。核心算法（BOM 递归展开、Process Group、依赖关系计算）封装为独立服务模块，便于单元测试和复用。

**Tech Stack:** Vue 3, Element Plus, TypeScript, Spring Boot, MySQL, Redis

```java
package com.erp.production.service;

import com.erp.production.domain.BOMNode;
import com.erp.production.repository.BOMRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

import java.util.*;

@Service
@RequiredArgsConstructor
public class BOMExpandService {
    
    private final BOMRepository bomRepository;
    private static final int MAX_LEVEL = 10;
    private static final int MAX_NODES = 1000;
    
    public BOMNode expandBOM(String bomId, double parentQuantity) {
        Set<String> visited = new HashSet<>();
        BOMNode root = bomRepository.findById(bomId)
            .orElseThrow(() -> new RuntimeException("BOM not found: " + bomId));
        
        expandNode(root, parentQuantity, 0, visited, new AtomicInteger(0));
        return root;
    }
    
    private void expandNode(BOMNode node, double parentQty, int level, 
                           Set<String> visited, AtomicInteger nodeCount) {
        // 循环检测
        if (visited.contains(node.getPartNumber())) {
            throw new BOMCycleException("BOM cycle detected at: " + node.getPartNumber());
        }
        
        // 层级限制
        if (level >= MAX_LEVEL) {
            throw new BOMDepthException("BOM exceeds max level: " + MAX_LEVEL);
        }
        
        // 节点数量限制
        if (nodeCount.incrementAndGet() > MAX_NODES) {
            throw new BOMSizeException("BOM exceeds max nodes: " + MAX_NODES);
        }
        
        node.setLevel(level);
        node.setQuantity(parentQty * node.getQuantity());
        
        List<BOMNode> children = bomRepository.findChildren(node.getNodeId());
        if (children == null || children.isEmpty()) {
            node.setIsLeaf(true);
            return;
        }
        
        node.setIsLeaf(false);
        visited.add(node.getPartNumber());
        
        for (BOMNode child : children) {
            expandNode(child, node.getQuantity(), level + 1, new HashSet<>(visited), nodeCount);
        }
        
        node.setChildren(children);
    }
}
```

```java
package com.erp.production.service;

import com.erp.production.domain.Recipe;
import com.erp.production.repository.RecipeRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

import java.util.Comparator;
import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class RecipeMatchService {
    
    private final RecipeRepository recipeRepository;
    
    public List<Recipe> findMatchingRecipes(String partNumber) {
        return recipeRepository.findByApplicablePartNumber(partNumber)
            .stream()
            .filter(r -> "active".equals(r.getStatus()))
            .sorted(Comparator
                .comparing(Recipe::getUsageCount, Comparator.reverseOrder())
                .thenComparing(Recipe::getCreatedAt, Comparator.reverseOrder()))
            .collect(Collectors.toList());
    }
    
    public Recipe recommendRecipe(String partNumber) {
        List<Recipe> matches = findMatchingRecipes(partNumber);
        return matches.isEmpty() ? null : matches.get(0);
    }
}
```

```java
package com.erp.production.service;

import com.erp.production.domain.BOMNode;
import com.erp.production.domain.Process;
import org.springframework.stereotype.Service;

import java.util.*;

@Service
public class ProcessGroupService {
    
    public List<Process> groupProcesses(BOMNode root, String jobId) {
        List<Process> processes = new ArrayList<>();
        Map<String, Process> teamProcessMap = new HashMap<>();
        Map<String, String> nodeToProcessMap = new HashMap<>();
        
        // 后序遍历，从叶子向上合并
        traverseAndGroup(root, jobId, processes, teamProcessMap, nodeToProcessMap);
        
        // 计算依赖关系
        calculateDependencies(root, nodeToProcessMap, processes);
        
        // 重命名 Process
        renameProcesses(processes);
        
        return processes;
    }
    
    private void traverseAndGroup(BOMNode node, String jobId,
                                   List<Process> processes,
                                   Map<String, Process> teamProcessMap,
                                   Map<String, String> nodeToProcessMap) {
        if (node.getChildren() != null) {
            for (BOMNode child : node.getChildren()) {
                traverseAndGroup(child, jobId, processes, teamProcessMap, nodeToProcessMap);
            }
        }
        
        if (node.getSelectedRecipe() == null) return;
        
        String teamId = node.getSelectedRecipe().getTeamId();
        String teamName = node.getSelectedRecipe().getTeamName();
        
        // 检查父节点是否同 Team
        String parentProcessId = node.getParentNodeId() != null ? 
            nodeToProcessMap.get(node.getParentNodeId()) : null;
        
        Process process;
        if (parentProcessId != null) {
            Process parentProcess = processes.stream()
                .filter(p -> p.getProcessId().equals(parentProcessId))
                .findFirst()
                .orElse(null);
            
            if (parentProcess != null && parentProcess.getTeamId().equals(teamId)) {
                // 合并到父 Process
                process = parentProcess;
            } else {
                process = createOrGetProcess(teamId, teamName, jobId, processes, teamProcessMap);
            }
        } else {
            process = createOrGetProcess(teamId, teamName, jobId, processes, teamProcessMap);
        }
        
        process.getNodes().add(node);
        node.setProcessGroupId(process.getProcessId());
        nodeToProcessMap.put(node.getNodeId(), process.getProcessId());
    }
    
    private Process createOrGetProcess(String teamId, String teamName, String jobId,
                                        List<Process> processes,
                                        Map<String, Process> teamProcessMap) {
        String key = jobId + "_" + teamId;
        if (teamProcessMap.containsKey(key)) {
            return teamProcessMap.get(key);
        }
        
        Process process = new Process();
        process.setProcessId(UUID.randomUUID().toString());
        process.setTeamId(teamId);
        process.setTeamName(teamName);
        process.setJobId(jobId);
        process.setNodes(new ArrayList<>());
        process.setPredecessorIds(new ArrayList<>());
        process.setStatus("pending");
        
        processes.add(process);
        teamProcessMap.put(key, process);
        return process;
    }
    
    private void calculateDependencies(BOMNode node, Map<String, String> nodeToProcessMap,
                                        List<Process> processes) {
        // 父子依赖：子节点所在 Process 依赖父节点所在 Process
        if (node.getChildren() != null) {
            String parentProcessId = nodeToProcessMap.get(node.getNodeId());
            for (BOMNode child : node.getChildren()) {
                String childProcessId = nodeToProcessMap.get(child.getNodeId());
                if (childProcessId != null && !childProcessId.equals(parentProcessId)) {
                    Process childProcess = processes.stream()
                        .filter(p -> p.getProcessId().equals(childProcessId))
                        .findFirst()
                        .orElse(null);
                    if (childProcess != null && !childProcess.getPredecessorIds().contains(parentProcessId)) {
                        childProcess.getPredecessorIds().add(parentProcessId);
                    }
                }
                calculateDependencies(child, nodeToProcessMap, processes);
            }
        }
    }
    
    private void renameProcesses(List<Process> processes) {
        Map<String, Integer> teamCount = new HashMap<>();
        for (Process process : processes) {
            String teamName = process.getTeamName();
            int count = teamCount.getOrDefault(teamName, 0);
            process.setProcessName(teamName + "-" + (count + 1));
            teamCount.put(teamName, count + 1);
        }
    }
}
```

```java
package com.erp.production.controller;

import com.erp.production.dto.*;
import com.erp.production.service.*;
import lombok.RequiredArgsConstructor;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/production/job-config")
@RequiredArgsConstructor
public class JobConfigController {
    
    private final JobConfigService jobConfigService;
    private final BOMExpandService bomExpandService;
    private final RecipeMatchService recipeMatchService;
    private final ProcessGroupService processGroupService;
    
    @PostMapping("/jobs")
    public ResponseEntity<Job> createJob(@RequestBody JobCreateRequest request) {
        Job job = jobConfigService.createJob(request);
        return ResponseEntity.ok(job);
    }
    
    @GetMapping("/bom/{bomId}/expand")
    public ResponseEntity<BOMExpandResponse> expandBOM(
            @PathVariable String bomId,
            @RequestParam Double quantity) {
        BOMNode root = bomExpandService.expandBOM(bomId, quantity);
        BOMExpandResponse response = new BOMExpandResponse();
        response.setRoot(root);
        // 计算 totalNodes 和 maxLevel
        return ResponseEntity.ok(response);
    }
    
    @GetMapping("/recipes/match")
    public ResponseEntity<RecipeMatchResponse> matchRecipes(
            @RequestParam String partNumber) {
        List<Recipe> recipes = recipeMatchService.findMatchingRecipes(partNumber);
        Recipe recommended = recipeMatchService.recommendRecipe(partNumber);
        
        RecipeMatchResponse response = new RecipeMatchResponse();
        response.setRecipes(recipes);
        response.setRecommended(recommended);
        return ResponseEntity.ok(response);
    }
    
    @PostMapping("/processes/group")
    public ResponseEntity<ProcessGroupResponse> groupProcesses(
            @RequestBody BOMNode root,
            @RequestParam String jobId) {
        List<Process> processes = processGroupService.groupProcesses(root, jobId);
        // 构建依赖图
        ProcessGroupResponse response = new ProcessGroupResponse();
        response.setProcesses(processes);
        return ResponseEntity.ok(response);
    }
    
    @PostMapping("/jobs/{jobId}/submit")
    public ResponseEntity<Job> submitJob(@PathVariable String jobId) {
        Job job = jobConfigService.submitJob(jobId);
        return ResponseEntity.ok(job);
    }
}
```

```typescript
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { Job, BOMNode, Recipe, Process } from '@/views/production/JobConfig/types/jobConfig';
import { jobConfigApi } from '@/api/production/job';

export const useJobConfigStore = defineStore('jobConfig', () => {
  // State
  const currentJob = ref<Job | null>(null);
  const bomTree = ref<BOMNode | null>(null);
  const processes = ref<Process[]>([]);
  const isLoading = ref(false);
  const error = ref<string | null>(null);
  
  // Getters
  const configuredNodes = computed(() => {
    const nodes: BOMNode[] = [];
    const traverse = (node: BOMNode) => {
      if (node.selectedRecipe) nodes.push(node);
      node.children?.forEach(traverse);
    };
    if (bomTree.value) traverse(bomTree.value);
    return nodes;
  });
  
  const unconfiguredNodes = computed(() => {
    const nodes: BOMNode[] = [];
    const traverse = (node: BOMNode) => {
      if (!node.selectedRecipe) nodes.push(node);
      node.children?.forEach(traverse);
    };
    if (bomTree.value) traverse(bomTree.value);
    return nodes;
  });
  
  const isComplete = computed(() => unconfiguredNodes.value.length === 0);
  
  // Actions
  async function createJob(data: Parameters<typeof jobConfigApi.createJob>[0]) {
    isLoading.value = true;
    try {
      const res = await jobConfigApi.createJob(data);
      currentJob.value = res.data;
      return res.data;
    } catch (e) {
      error.value = e instanceof Error ? e.message : '创建 Job 失败';
      throw e;
    } finally {
      isLoading.value = false;
    }
  }
  
  async function expandBOM(bomId: string, quantity: number) {
    isLoading.value = true;
    try {
      const res = await jobConfigApi.expandBOM(bomId, quantity);
      bomTree.value = res.data;
      return res.data;
    } catch (e) {
      error.value = e instanceof Error ? e.message : '展开 BOM 失败';
      throw e;
    } finally {
      isLoading.value = false;
    }
  }
  
  async function selectRecipe(nodeId: string, recipe: Recipe) {
    const updateNode = (node: BOMNode): boolean => {
      if (node.nodeId === nodeId) {
        node.selectedRecipe = recipe;
        node.status = 'manual';
        return true;
      }
      return node.children?.some(updateNode) || false;
    };
    
    if (bomTree.value) {
      updateNode(bomTree.value);
    }
  }
  
  async function calculateProcesses() {
    if (!bomTree.value || !currentJob.value) return;
    
    isLoading.value = true;
    try {
      const res = await jobConfigApi.groupProcesses(bomTree.value, currentJob.value.jobId);
      processes.value = res.data.processes;
    } catch (e) {
      error.value = e instanceof Error ? e.message : '计算 Process 失败';
      throw e;
    } finally {
      isLoading.value = false;
    }
  }
  
  function reset() {
    currentJob.value = null;
    bomTree.value = null;
    processes.value = [];
    error.value = null;
  }
  
  return {
    currentJob,
    bomTree,
    processes,
    isLoading,
    error,
    configuredNodes,
    unconfiguredNodes,
    isComplete,
    createJob,
    expandBOM,
    selectRecipe,
    calculateProcesses,
    reset,
  };
});
```

```vue
<script setup lang="ts">
import { ref, reactive, computed } from 'vue';
import { useRouter } from 'vue-router';
import { useJobConfigStore } from '@/stores/jobConfig';
import { ElMessage, ElMessageBox } from 'element-plus';
import JobBasicInfo from './components/JobBasicInfo.vue';
import BOMTree from './components/BOMTree.vue';
import ProcessPreview from './components/ProcessPreview.vue';

const router = useRouter();
const store = useJobConfigStore();

const currentStep = ref(0);
const basicInfo = reactive({
  partNumber: '',
  bomId: '',
  quantity: 1,
});

const nodeCount = computed(() => {
  let count = 0;
  const traverse = (node: any) => {
    count++;
    node.children?.forEach(traverse);
  };
  if (store.bomTree) traverse(store.bomTree);
  return count;
});

const maxLevel = computed(() => {
  let max = 0;
  const traverse = (node: any, level: number) => {
    max = Math.max(max, level);
    node.children?.forEach((child: any) => traverse(child, level + 1));
  };
  if (store.bomTree) traverse(store.bomTree, 0);
  return max;
});

async function onPartChange(partNumber: string) {
  basicInfo.partNumber = partNumber;
}

async function onBOMChange(bomId: string) {
  basicInfo.bomId = bomId;
  if (bomId) {
    try {
      await store.expandBOM(bomId, basicInfo.quantity);
      currentStep.value = 0;
    } catch (e) {
      ElMessage.error('BOM 展开失败');
    }
  }
}

function onQuantityChange(quantity: number) {
  basicInfo.quantity = quantity;
}

async function saveDraft() {
  // TODO: 调用保存草稿 API

    await ElMessageBox.confirm('确认提交 Job 进入排程？', '提示', { type: 'warning' });
    // TODO: 调用提交 API
    ElMessage.success('Job 已提交，进入待排程状态');
    router.push('/production/jobs');
  } catch {
    // 取消
  }
}

function goBack() {
  router.back();
}
```
